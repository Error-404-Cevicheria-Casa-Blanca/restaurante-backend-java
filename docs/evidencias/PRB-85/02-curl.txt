# 1. Login exitoso con terminal_codigo (POST /api/v1/auth/login)
curl -X POST http://localhost:8080/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{
    "usuario": "mozo_01",
    "clave": "password123",
    "terminal_codigo": "TERM-01"
  }'
